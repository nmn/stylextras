### Example code

```tsx
'use client';

import { ButtonGroup } from '../button-group';
import { DialogCommandBridge } from '../dialog/client';
import { Field, FieldLabel } from '../field';
import { Input } from '../input';
import {
  Sheet,
  SheetBody,
  SheetClose,
  SheetFooter,
  SheetHeader,
  type SheetSide,
  SheetTrigger,
} from './index';

function ProjectSettingsSheet({ id, side }: { id: string; side: SheetSide }) {
  return (
    <Sheet
      id={id}
      side={side}
      aria-labelledby={`${id}-title`}
      aria-describedby={`${id}-description`}
    >
      <SheetHeader>
        <h2 id={`${id}-title`}>Project settings</h2>
      </SheetHeader>
      <SheetBody>
        <p id={`${id}-description`}>Update the project and team names.</p>
        <Field>
          <FieldLabel htmlFor={`${id}-project-name`}>Project name</FieldLabel>
          <Input id={`${id}-project-name`} defaultValue="StyleXtras" />
        </Field>
        <Field>
          <FieldLabel htmlFor={`${id}-team-name`}>Team</FieldLabel>
          <Input id={`${id}-team-name`} defaultValue="Interface systems" />
        </Field>
      </SheetBody>
      <SheetFooter>
        <ButtonGroup variant="actions" aria-label="Project settings actions">
          <SheetClose target={id}>Cancel</SheetClose>
          <SheetClose target={id} variant="primary">
            Save
          </SheetClose>
        </ButtonGroup>
      </SheetFooter>
      <DialogCommandBridge target={id} />
    </Sheet>
  );
}

export default function Example() {
  return (
    <>
      <SheetTrigger target="end-settings-sheet">Open end sheet</SheetTrigger>
      <ProjectSettingsSheet id="end-settings-sheet" side="end" />

      <SheetTrigger target="start-settings-sheet">
        Open start sheet
      </SheetTrigger>
      <ProjectSettingsSheet id="start-settings-sheet" side="start" />
    </>
  );
}
```

## Anatomy

- `Sheet` — Styled `<dialog>`.
- `SheetBody` — Styled `<div>`.
- `SheetClose` — Styled `<button>`.
- `SheetFooter` — Styled `<div>`.
- `SheetHeader` — Styled `<div>`.
- `SheetTrigger` — Styled `<button>`.
- `DialogCommandBridge` — Optional feature-detected invoker-command fallback.

## Accessibility and browser behavior

Each part renders and styles the native element named above. The API reference states the exact React or inherited prop base, including whether `ref` is accepted, every omitted native prop, and every component-applied default. Any native prop not named in `Omit` remains accepted, including standard ARIA/data attributes and event handlers. `sx` is composed last.

Give the native dialog a heading relationship and associate concise descriptive
copy when it helps explain the task. The example installs
`DialogCommandBridge`, which feature-detects native invoker commands before
loading its fallback. `LazySheet` installs the same fallback internally.

## API reference

### Sheet

`SheetProps`

Complete native/inherited prop base: `Omit<React.ComponentPropsWithRef<"dialog">, "aria-label" | "aria-labelledby" | "className" | "style">` plus one required accessible name.

- `aria-label` or `aria-labelledby` — Required accessible dialog name.
- `side` — `SheetSide`. Default: `'end'`.
- `sx` — `stylex.StyleXStyles`.

### SheetBody

`SheetBodyProps`

Complete native/inherited prop base: `Omit<React.ComponentPropsWithRef<"div">, "className" | "style">`. The props below complete `SheetBodyProps`.

- `sx` — `stylex.StyleXStyles`.

### SheetClose

`SheetCloseProps`

Complete native/inherited prop base: `SheetTriggerProps`. The props below complete `SheetCloseProps`.

- `size` — `ButtonSize`. Default: `'md'`.
- `sx` — `stylex.StyleXStyles`.
- `target` — `string`. Required.
- `type` — `SheetCloseProps["type"]`. Default: `'button'`.
- `variant` — `ButtonVariant`. Default: `'outline'`.
